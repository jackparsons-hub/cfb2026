# ==============================================================================
# CFB2026 automation -- crontab entries
# Install with: crontab -e   (paste below, adjusting paths/Rscript location)
# Assumes CFBD_API_KEY and PG* env vars are set in a sourced profile, e.g.
# via `env` prefix or a wrapper shell script that sources ~/.cfb2026_env first.
# ==============================================================================

# Daily overnight pull: refresh schedule/scores/box-scores/ratings/lines for
# the current season. Runs at 3:00 AM local -- after West Coast Saturday/late
# games are final and CFBD has ingested updated stats.
0 3 * * * cd /opt/cfb2026 && /usr/bin/Rscript R/01_ingest_cfbfastr.R >> logs/ingest_$(date +\%Y\%m\%d).log 2>&1

# Tuesday 6:00 AM: betting lines tend to stabilize/open for the coming week by
# Sunday night-Monday; grab an early-week snapshot in addition to the nightly pull.
0 6 * * 2 cd /opt/cfb2026 && /usr/bin/Rscript -e 'source("R/01_ingest_cfbfastr.R")' >> logs/lines_tue_$(date +\%Y\%m\%d).log 2>&1

# Wednesday 4:00 AM: rebuild engineered features (rolling form, rest days,
# rating diffs) now that Tuesday's line move + Monday night results are in.
0 4 * * 3 cd /opt/cfb2026 && /usr/bin/Rscript R/03_feature_engineering.R >> logs/features_$(date +\%Y\%m\%d).log 2>&1

# Wednesday 5:00 AM: weekly model retrain + walk-forward re-validation, so the
# champion model reflects the latest completed week before Thursday kickoffs.
0 5 * * 3 cd /opt/cfb2026 && /usr/bin/Rscript R/04_model_comparison.R >> logs/train_$(date +\%Y\%m\%d).log 2>&1

# Wednesday 6:30 AM: generate this week's pairwise predictions + Vegas/FPI/Elo
# comparison report, ahead of the Thursday night game.
# NOTE: TARGET_WEEK is a fixed placeholder below -- either bump it manually
# each week or (better) compute it dynamically inside 05_predict_and_compare.R
# from today's date against the season's Week 0 start date.
30 6 * * 3 cd /opt/cfb2026 && CHAMPION_MODEL=gru TARGET_WEEK=1 /usr/bin/Rscript R/05_predict_and_compare.R >> logs/predict_$(date +\%Y\%m\%d).log 2>&1

# Sunday 11:00 PM: final catch-up pull once the full week's games (through
# Sunday MAC/G5 games) are complete, closing out the week's data for backtesting.
0 23 * * 0 cd /opt/cfb2026 && /usr/bin/Rscript R/01_ingest_cfbfastr.R >> logs/ingest_sun_$(date +\%Y\%m\%d).log 2>&1
